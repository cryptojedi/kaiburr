% \heat{<value>}{<table max>}{<cell text>}: 4-step orange heat map,
% square-root scale so small values stay distinguishable
\newcommand{\heat}[3]{%
  \pgfmathsetmacro\heatt{3*sqrt(#1/#2)}%
  \pgfmathtruncatemacro\heatseg{min(int(\heatt),2)}%
  \pgfmathtruncatemacro\heatpct{100*(\heatt-\heatseg)}%
  \pgfmathtruncatemacro\heatnext{\heatseg+1}%
  \edef\heatcmd{\noexpand\cellcolor{heat\heatnext!\heatpct!heat\heatseg}}%
  \heatcmd#3}
